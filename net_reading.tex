% Allow column/page breaks between chapter heading and first verse.
% The scripture package puts \nobreak + \@afterheading after the heading
% paragraph, gluing heading to verse 1.  With large lettrines in a
% two-column layout this forces the whole block to the next column,
% leaving large blank gaps.  \allowchapbreak cancels both mechanisms.
\makeatletter

\makeatother

% \bbook[image]{Main Title}{Subtitle}{Argument text}{book-dir}
% Chapter table (if any) must be set via \gdef\bbookchaptable{...} BEFORE calling \bbook.
\NewDocumentCommand{\bbook}{O{} m m m m}{%
  \clearpage
  \gdef\currentbookdir{#5}%
  \thispagestyle{scrheadings}%
  \hypertarget{book-#5}{}%
  \IfBlankTF{#1}{%
    \savebox{\bbookimgbox}{}%
  }{%
    \savebox{\bbookimgbox}{\makebox[\textwidth][c]{\includegraphics[width=\textwidth]{#1}}}%
  }%
  \twocolumn[%
    \usebox{\bbookimgbox}\par\nobreak
    \vspace*{-10pt}%
    {\centering\huge\booktitlefont\scshape #2%
      \IfBlankF{#3}{,\\\large #3}%
      \\\char"2766\par}%
    \IfBlankF{#4}{%
      {\centering\booktitlefont\textsc{the argument.}\par}%
      \vspace{2pt}%
      {\centering\parbox{\dimexpr\textwidth-2em}{\itshape\scriptsize #4\par}\par}%
      \vspace{5pt}%
    }%
    \bbookchaptable
    \vspace{5pt}%
  ]%
  \gdef\bbookchaptable{}
}
