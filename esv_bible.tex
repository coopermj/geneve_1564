% Baseline grid unit: 11pt body with \linespread{0.9} gives 12.23991pt leading.
% 0.9\dimexpr13.6pt uses the same factor*dimen truncation LaTeX applies when
% computing \baselineskip, so \gridunit and \baselineskip are sp-identical.
% Every vertical interval in the body (heading blocks, poetry skips, chapter
% paras, book-opener padding) must be a whole multiple of \gridunit so both
% columns stay in line-for-line register.
\newlength{\gridunit}
\setlength{\gridunit}{0.9\dimexpr13.6pt\relax}

% slack that \flushbottom stretched into the interline glue on every page.
% Pin the text height to exactly 46 baselines (\topskip + 45\gridunit =
% 561.80pt, +0.64pt into the bottom margin) so flushbottom never stretches.
\setlength{\textheight}{\dimexpr\topskip+45\gridunit\relax}

% Heading-orphan guard, list-penalty leg: scripture sets
% \@endparpenalty=\@lowpenalty (51) and \@beginparpenalty=-51 for its inner
% list environments; the list END/BEGIN at a poetry chapter boundary inserts
% those penalty nodes AFTER any \nobreak we emit (consecutive penalties do
% not merge — each one is its own breakpoint), so a heading could still be
% orphaned at a column foot. \headingkeep neutralizes them (globally — the
% insertion happens inside \ch, deeper in grouping) right after a heading;

\NewDocumentCommand{\bbook}{O{} m m m m}{%
  \clearpage
  \gdef\currentbookdir{#5}%
  \thispagestyle{scrheadings}%
  \hypertarget{book-#5}{}%
  \IfBlankTF{#1}{%
    \savebox{\bbookimgbox}{}%
  }{%
    \savebox{\bbookimgbox}{\makebox[\textwidth][c]{\includegraphics[width=\textwidth]{#1}}}%
  }%
  \begin{lrbox}{\bbookheadbox}\parbox{\textwidth}{%
    \usebox{\bbookimgbox}\par\nobreak
    \vspace*{-10pt}%
    {\centering\huge\booktitlefont\scshape #2%
      \IfBlankF{#3}{,\\\large #3}%
      \\\char"2766\par}%
    \IfBlankF{#4}{%
      {\centering\booktitlefont\textsc{the argument.}\par}%
      \vspace{2pt}%
      {\centering\parbox{\dimexpr\textwidth-2em}{\itshape\scriptsize #4\par}\par}%
      \vspace{5pt}%
    }%
    \bbookchaptable
    \vspace{5pt}%
  }\end{lrbox}%
  \setlength{\bbookheadnat}{\dimexpr\ht\bbookheadbox+\dp\bbookheadbox\relax}%
  \twocolumn[{%
    \usebox{\bbookheadbox}\par
    % pad to ceil(natural/\gridunit) grid lines (ceil via rounding division)
    \vspace{\dimexpr\numexpr(2*\bbookheadnat+\gridunit-2)/(2*\gridunit)\relax\gridunit-\bbookheadnat\relax}%
  }]%
  \gdef\bbookchaptable{}
}
