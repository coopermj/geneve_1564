%% line_breaking.tex — narrow-measure line breaking (shared by all editions)
%%
%% Section 3 of docs/typography-recommendations.md. The two-column measure runs
%% ~40–45 characters; TeX's defaults there produce loose lines and hyphen
%% ladders. Input this AFTER \usepackage[english]{babel} (so hyphenation
%% patterns and the current language are established) and before \begin{document}.

% --- Breaking parameters -----------------------------------------------------
% Slightly looser badness tolerance + a small emergency stretch lets TeX solve
% tight lines without overfull boxes; the demerit bumps discourage stacked and
% final-line hyphens (Oxford/Cambridge house practice in a narrow column).
% \LBbaseline (define before input) disables the whole Section-3 change for
% A/B measurement against the lua-typo diagnostic; production leaves it unset.
\ifdefined\LBbaseline\else
  \tolerance=400
  \emergencystretch=.5em
  \doublehyphendemerits=7500
  \finalhyphendemerits=7500
\fi

% --- lua-typo diagnostic (opt-in) -------------------------------------------
% Colours widows/orphans/hyphen ladders/short lines and logs them. NOT loaded
% in normal builds. Enable with:
%   lualatex "\def\enabletypocheck{}\input{esv_bible.tex}"
% Focused checks (NOT [All]): in a ~40-char justified column, ShortLines /
% ShortPages / EOLShortWords fire on nearly every page and drown the signal.
% Keep the ones Section 3 actually targets — hyphen ladders, paragraph-final
% and end-of-page hyphens — plus widows/orphans (residuals lua-widow-control
% couldn't resolve).
\ifdefined\enabletypocheck
  \IfFileExists{lua-typo.sty}
    {\usepackage[Widows,Orphans,EOPHyphens,RepeatedHyphens,ParLastHyphen]{lua-typo}}
    {\PackageWarning{line_breaking}{\string\enabletypocheck\space set but
       lua-typo is not installed; skipping diagnostic}}
\fi
